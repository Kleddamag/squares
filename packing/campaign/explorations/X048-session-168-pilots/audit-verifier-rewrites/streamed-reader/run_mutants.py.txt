"""Mutants of the streamed kernel verifier's new code, and which checks catch each.

Each mutant is one or more textual patches of the verifier at the worktree, each asserted
to apply exactly once, written to a scratch copy. Against each mutant run:

- `old`: the committed tests of 601bbf110, `tests/test_verify_n17_certificates.py`, whole;
- `new`: this review's `tests/test_verify_n17_node_stream.py`, whole (slow test included);
- `diff`: the differential harness `nodestream_differential.py` at 300 cases.

All three bind the mutant through `mutant_plugin` (R8_MUTANT). A mutant is CAUGHT by a
check when that check fails on it. Prints one JSON line per mutant.

Usage, from the worktree's `packing/` with the worktree on PYTHONPATH:
    python run_mutants.py SCRATCH_DIR [NAME ...]
"""

def run(command: list[str], env: dict[str, str], timeout: int) -> tuple[bool, str]:
    try:
        done = subprocess.run(
            command, cwd=PACKING, env=env, capture_output=True, text=True, timeout=timeout,
            check=False,
        )
    except subprocess.TimeoutExpired:
        return False, "timeout"
    tail = (done.stdout + done.stderr).strip().splitlines()[-3:]
    return done.returncode == 0, " | ".join(tail)[-400:]


def main() -> int:
    scratch = Path(sys.argv[1])
    scratch.mkdir(parents=True, exist_ok=True)
    only = set(sys.argv[2:])
    source = COMMITTED.read_text()
    base = {**os.environ, "PYTHONPATH": f"{HERE}:{PACKING / 'src'}:{PACKING}"}
    for name, weakens, patches in MUTANTS:
        if only and name not in only:
            continue
        text = source
        for old, new in patches:
            assert text.count(old) == 1, (name, old)
            text = text.replace(old, new)
        path = scratch / f"{name}.py"
        _ = path.write_text(text)
        env = {**base, "R8_MUTANT": str(path)}
        started = time.time()
        results: dict[str, object] = {}
        for check, command, timeout in (
            ("old", [PYTHON, "-m", "pytest", "-q", "-x", "-p", "no:cacheprovider", "-p",
                     "mutant_plugin", "tests/test_verify_n17_certificates.py"], 900),
            ("new", [PYTHON, "-m", "pytest", "-q", "-x", "-p", "no:cacheprovider", "-p",
                     "mutant_plugin", "tests/test_verify_n17_node_stream.py"], 900),
            ("diff", [PYTHON, "-c", "import mutant_plugin, runpy, sys; sys.argv = ['h', '300', "
                      f"'11', '{scratch / (name + '.json.gz')}']; "
                      f"runpy.run_path('{HERE / 'nodestream_differential.py'}', "
                      "run_name='__main__')"], 900),
        ):
            passed, tail = run(command, env, timeout)
            results[check] = {"caught": not passed, "tail": tail}
        print(json.dumps({"mutant": name, "weakens": weakens, **results,
                          "seconds": round(time.time() - started, 1)}), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
